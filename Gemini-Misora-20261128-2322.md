### B. Skrip Audit AST Lokal Tanpa Install Library (`scan_static_ast.py`)

Simpan skrip berikut di **`D:\Reference\Misora-Reference\audit_post_t04\scan_static_ast.py`** lalu jalankan dengan `py -3.12 -B`. Skrip ini mendeteksi 4 sumber bug & *dead code* paling berbahaya di file monolitik 24.000 baris sambil menyaring *false positive* Tkinter (`bind`, `command`, `after`, `getattr`, `tests/`):

1. **Duplicate method/function shadowing** (dua metode bernama sama di dalam kelas yang sama — definisi pertama diam-diam tertimpa).
2. **Unreachable code** (kode setelah `return`, `raise`, `break`, atau `continue` pada blok yang sama).
3. **Mutable default arguments** (`def fn(x=[])` / `x={}`).
4. **Kandidat fungsi/metode tak terpakai** yang benar-benar `0` kemunculan nama (baik sebagai *identifier* maupun *string literal*) di seluruh `misora.py`, `core/`, `assets/`, `tests/`, dan `tools/`.

`​`​`python
# D:\Reference\Misora-Reference\audit_post_t04\scan_static_ast.py
import ast
import pathlib
import re
from collections import Counter

REPO = pathlib.Path(r"D:\Misora-Editor")
OUT_FILE = pathlib.Path(r"D:\Reference\Misora-Reference\audit_post_t04\01_ast_static_report.md")

PROD_FILES = [
    REPO / "misora.py",
    *sorted((REPO / "core").glob("*.py")),
    *sorted((REPO / "assets").glob("*.py")),
]
ALL_SEARCH_FILES = [
    *PROD_FILES,
    REPO / "__init__.py",
    *sorted((REPO / "tests").glob("*.py")),
    *sorted((REPO / "tools").glob("*.py")),
]

TOKEN_RE = re.compile(r"[A-Za-z_][A-Za-z0-9_]*")

def read_text_bom_safe(p: pathlib.Path) -> str:
    raw = p.read_bytes()
    return raw.removeprefix(bytes([239, 187, 191])).decode("utf-8")

def check_unreachable(stmts, rel, out):
    for i, s in enumerate(stmts):
        if isinstance(s, (ast.Return, ast.Raise, ast.Break, ast.Continue)) and i + 1 < len(stmts):
            nxt = stmts[i + 1]
            out.append(f"- `{rel}:{nxt.lineno}` — Unreachable `{type(nxt).__name__}` after `{type(s).__name__}` at L{s.lineno}")
            break
        for child_body_attr in ("body", "orelse", "finalbody"):
            sub = getattr(s, child_body_attr, None)
            if isinstance(sub, list) and sub and isinstance(sub[0], ast.stmt):
                check_unreachable(sub, rel, out)
        if isinstance(s, ast.Try):
            for h in s.handlers:
                check_unreachable(h.body, rel, out)

def main():
    global_tokens = Counter()
    for p in ALL_SEARCH_FILES:
        if p.exists():
            global_tokens.update(TOKEN_RE.findall(read_text_bom_safe(p)))

    dups, unreach, mutables, dead_candidates = [], [], [], []

    for p in PROD_FILES:
        if not p.exists():
            continue
        rel = p.relative_to(REPO).as_posix()
        src = read_text_bom_safe(p)
        tree = ast.parse(src, filename=str(p))

        # 1. Unreachable code
        check_unreachable(tree.body, rel, unreach)

        # 2. Duplicate top-level defs & class methods
        top_seen = {}
        for node in tree.body:
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                if node.name in top_seen:
                    dups.append(f"- `{rel}:{node.lineno}` — Duplicate top-level `{node.name}` (first at L{top_seen[node.name]})")
                top_seen[node.name] = node.lineno
                if not node.name.startswith("__") and global_tokens[node.name] <= 1:
                    dead_candidates.append(f"- `{rel}:{node.lineno}` — Top-level `def {node.name}()` (global token hits: {global_tokens[node.name]})")
            elif isinstance(node, ast.ClassDef):
                method_seen = {}
                for item in node.body:
                    if isinstance(item, (ast.FunctionDef, ast.AsyncFunctionDef)):
                        if item.name in method_seen:
                            dups.append(f"- `{rel}:{item.lineno}` — Duplicate method `{node.name}.{item.name}` (first at L{method_seen[item.name]})")
                        method_seen[item.name] = item.lineno
                        if not (item.name.startswith("__") and item.name.endswith("__")) and global_tokens[item.name] <= 1:
                            dead_candidates.append(f"- `{rel}:{item.lineno}` — Method `{node.name}.{item.name}()` (global token hits: {global_tokens[item.name]})")

        # 3. Mutable defaults
        for node in ast.walk(tree):
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                for d in (*node.args.defaults, *(x for x in node.args.kw_defaults if x is not None)):
                    if isinstance(d, (ast.List, ast.Dict, ast.Set)):
                        mutables.append(f"- `{rel}:{node.lineno}` — Mutable default `{type(d).__name__}` in `{node.name}()`")

    lines = [
        "# Post-T-04 Local AST Static Scan (Read-Only, Zero-.pyc)",
        "",
        f"## 1. Duplicate Method / Function Shadowing ({len(dups)})",
        *(dups or ["- None"]),
        "",
        f"## 2. Unreachable Code ({len(unreach)})",
        *(unreach or ["- None"]),
        "",
        f"## 3. Mutable Default Arguments ({len(mutables)})",
        *(mutables or ["- None"]),
        "",
        f"## 4. Zero-Caller Dead-Code Candidates ({len(dead_candidates)})",
        "*(Verify manually against dynamic string assembly / Tk callbacks before ever deleting)*",
        *(dead_candidates or ["- None"]),
    ]
    OUT_FILE.write_text("\n".join(lines) + "\n", encoding="utf-8")
    print(f"WROTE {OUT_FILE} (dups={len(dups)}, unreach={len(unreach)}, mutables={len(mutables)}, dead={len(dead_candidates)})")

if __name__ == "__main__":
    main()
